14: /**
15:  * The design system, carried over from the web build.
16:  *
17:  * The rule that produced it: premium in this category is restraint plus
18:  * hierarchy, not decoration. One thing per screen is loud — the balance —
19:  * and everything else is whispered until you look at it directly. Money is
20:  * set large with tight tracking; labels shrink to uppercase micro-type with
21:  * wide tracking. The contrast between those two does most of the work.
22:  */

63: /** Money is always tabular — `tnum`, not a promise in a comment. Roboto's
64:  *  default figures are proportional, so a column of amounts whose digits do
65:  *  not line up is the fastest way for an app to look unfinished, and the
66:  *  ledger is a column of amounts. */
67: private const val TABULAR = "tnum"
68:

97: @Composable
98: fun SpendWiseTheme(
99:     dark: Boolean = isSystemInDarkTheme(),
100:     content: @Composable () -> Unit,
101: ) {
102:     MaterialTheme(
103:         colorScheme = if (dark) DarkScheme else LightScheme,
104:         typography = Typography(
105:             headlineLarge = TextStyle(fontSize = 26.sp, fontWeight = FontWeight.ExtraBold,
106:                 letterSpacing = (-0.6).sp),
107:             titleMedium = TextStyle(fontSize = 15.sp, fontWeight = FontWeight.SemiBold),
108:             bodyMedium = TextStyle(fontSize = 14.sp),
109:             bodySmall = TextStyle(fontSize = 12.sp),
110:             labelSmall = MicroLabel,
111:         ),
112:         content = content,
113:     )
